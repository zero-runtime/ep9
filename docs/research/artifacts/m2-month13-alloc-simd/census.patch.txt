+/// [`this_store_names`] in the escape mode of f-design L1: at a leave of `this`, mark
+/// `escaped` and go on. Returns (names, escaped).
+#[cfg(feature = "census")]
+fn this_store_names_esc(code: &[u8], names: &mut [u32; MAX_NAMES]) -> (usize, bool) {
+    let mut h = Holders::default();
+    let mut n = 0;
+    let mut esc = false;
+    for insn in Instructions::new(code) {
+        let Ok(insn) = insn else { break };
+        let op = insn.opcode;
+        let info = op.info();
+        match op {
+            Opcode::Ldar => {
+                h.acc = h.holds(insn.register(0));
+                continue;
+            }
+            Opcode::Star => {
+                if !h.set(insn.register(0), h.acc) {
+                    esc = true;
+                }
+                continue;
+            }
+            Opcode::Mov => {
+                let on = h.holds(insn.register(0));
+                if !h.set(insn.register(1), on) {
+                    esc = true;
+                }
+                continue;
+            }
+            Opcode::Return => continue,
+            Opcode::SetNamedProperty | Opcode::DefineNamedOwnProperty => {
+                if h.acc {
+                    esc = true;
+                }
+                if h.holds(insn.register(0)) {
+                    let k = insn.unsigned(1);
+                    if !names[..n].contains(&k) && n < MAX_NAMES {
+                        names[n] = k;
+                        n += 1;
+                    }
+                }
+                h.acc = false;
+                continue;
+            }
+            Opcode::GetNamedProperty => {
+                h.acc = false;
+                continue;
+            }
+            Opcode::GetNamedPropertyStar => {
+                h.acc = false;
+                let _ = h.set(insn.register(3), false);
+                continue;
+            }
+            _ => {}
+        }
+        if h.acc && info.acc.reads() {
+            esc = true;
+        }
+        let count = insn.operand_count();
+        let mut i = 0;
+        while i < count {
+            let k = insn.operand_kind(i);
+            match k {
+                OperandKind::Reg | OperandKind::RegInOut => {
+                    if h.holds(insn.register(i)) {
+                        esc = true;
+                    }
+                }
+                OperandKind::RegOut => {
+                    let _ = h.set(insn.register(i), false);
+                }
+                OperandKind::RegList | OperandKind::RegOutList => {
+                    let first = insn.register(i).to_operand();
+                    let len = insn.unsigned(i + 1) as i32;
+                    for j in 0..len {
+                        let r = Register::from_operand(first + j);
+                        if k == OperandKind::RegList && h.holds(r) {
+                            esc = true;
+                        }
+                        if k == OperandKind::RegOutList {
+                            let _ = h.set(r, false);
+                        }
+                    }
+                    i += 1;
+                }
+                OperandKind::RegPair | OperandKind::RegOutPair | OperandKind::RegOutTriple => {
+                    let first = insn.register(i).to_operand();
+                    let len = if k == OperandKind::RegOutTriple { 3 } else { 2 };
+                    for j in 0..len {
+                        let r = Register::from_operand(first + j);
+                        if k == OperandKind::RegPair && h.holds(r) {
+                            esc = true;
+                        }
+                        if k != OperandKind::RegPair {
+                            let _ = h.set(r, false);
+                        }
+                    }
+                }
+                _ => {}
+            }
+            i += 1;
+        }
+        if info.acc.writes() {
+            h.acc = false;
+        }
+    }
+    (n, esc)
+}

+/// The constructor chain of `proto` (f-design L1 (2)): walk the [[Prototype]] chain from
+/// `proto`; at each object read the own data property `constructor`; a bytecode constructor
+/// that is not derived adds its distinct names. Stops at a host function, a dictionary-mode
+/// object, an accessor, or 8 levels. Not a GC point.
+#[cfg(feature = "census")]
+pub(crate) fn census_chain(rt: &Runtime, proto: u64) -> Chain {
+    let mut ch = Chain {
+        name: String::from("(none)"),
+        bytecode: false,
+        names: Vec::new(),
+        escaped: false,
+        extra: 0,
+        levels: 0,
+        own_n: 0,
+    };
+    let ctor_key = rt.atom(crate::runtime::Atom::Constructor);
+    let mut cur = proto;
+    let mut first = true;
+    while ch.levels < 8 && kind::is_object(cur) {
+        let s = shape::shape_of(cur);
+        if shape::is_dictionary(s) {
+            break;
+        }
+        let Some((c, _)) = crate::props::own_get(cur, ctor_key) else {
+            break;
+        };
+        if !object::is_function(c) {
+            break;
+        }
+        let fi = object::function_info(c);
+        if first {
+            ch.name = fi.name.clone();
+        }
+        let need = fi_flags::BYTECODE | fi_flags::CONSTRUCTOR;
+        if fi.flags & need != need || fi.flags & fi_flags::DERIVED != 0 || fi.code.is_null() {
+            break;
+        }
+        if first {
+            ch.bytecode = true;
+        }
+        // SAFETY: as in `first_capacity`.
+        let code = unsafe { core::slice::from_raw_parts(fi.code, fi.code_len as usize) };
+        let mut names = [0u32; MAX_NAMES];
+        let (n, esc) = this_store_names_esc(code, &mut names);
+        ch.escaped |= esc;
+        for &k in &names[..n] {
+            // SAFETY: `k` is a verified String constant index of this function.
+            let key = unsafe { (*fi.consts.add(k as usize)).bits() };
+            if !ch.names.contains(&key) {
+                ch.names.push(key);
+            }
+        }
+        if first {
+            ch.own_n = ch.names.len();
+        }
+        ch.levels += 1;
+        first = false;
+        cur = shape::proto(s);
+    }
+    if ch.bytecode {
+        let ps = shape::shape_of(proto);
+        if !shape::is_dictionary(ps) {
+            ch.extra = (0..shape::count(ps))
+                .filter(|&i| is_proto_field(proto, ps, i) && !ch.names.contains(&shape::key_at(ps, i)))
+                .count();
+        }
+    }
+    ch
+}

+        });
+        c.lk_by_ctx.borrow_mut()[x][usize::from(found)] += 1;
+        c.lk_ctx_work.borrow_mut()[x] += work;
+        let b = if n <= 8 { n } else { n.next_power_of_two() };
+        *c.lk_by_n.borrow_mut().entry(b).or_insert(0) += 1;
+        {
+            let mut sites = c.lk_sites.borrow_mut();
+            let (_, sc) = sites
+                .entry(site as *const _ as usize)
+                .or_insert_with(|| (site, SiteCache::default()));
+            sc.calls += 1;
+            sc.work += work;
+            // One LRU list of 4 holds the answers of 1 and 2 entries too (its first 1 or 2).
+            let p = sc.lru.iter().position(|&y| y == e);
+            for (i, w) in WAYS.into_iter().enumerate() {
+                if p.is_some_and(|p| p < w) {
+                    sc.hit[i] += 1;
+                    sc.hit_work[i] += work;
+                }
+            }
+            lru(&mut sc.lru, 4, e);
+        }
+        let h = ((s >> 3) ^ key.wrapping_mul(0x9e37_79b9_7f4a_7c15).rotate_left(29)) as usize;
+        let mut g = c.lk_global.borrow_mut();
+        for (i, size) in GLOBAL.into_iter().enumerate() {
+            if g[i].is_empty() {
+                g[i] = vec![(0, 0); size];
+            }
+            let slot = &mut g[i][h & (size - 1)];
+            if *slot == e {
+                c.lk_global_hit.borrow_mut()[i] += 1;
+                c.lk_global_hit_work.borrow_mut()[i] += work;
+                c.lk_global_ctx_hit.borrow_mut()[i][x] += 1;
+            } else {
+                *slot = e;
+            }
+        }
+    });
+}

+fn key_name(k: u64) -> String {
+    if crate::kinds::is_string(k) {
+        crate::string::to_rust_string(k)
+    } else {
+        format!("<{k:#x}>")
+    }

+fn kind_name(k: usize) -> String {
+    let n = match k {
+        1 => "seq_latin1",
+        2 => "seq_utf16",
+        3 => "cons",
+        4 => "sliced",
+        5 => "thin",
+        8 => "symbol",
+        9 => "bigint",
+        16 => "object",
+        17 => "function",
+        18 => "array",
+        19 => "error",
+        20 => "iterator",
+        21 => "wrapper",
+        22 => "bound_function",
+        23 => "generator",
+        24 => "slot_object",
+        25 => "arguments",
+        64 => "props",
+        65 => "elements",
+        66 => "context",
+        67 => "accessor_pair",
+        68 => "feedback",
+        69 => "poly_ic",
+        70 => "ic_handler",
+        71 => "slots",
+        72 => "descriptors",
+        73 => "desc_index",
+        74 => "transitions",
+        75 => "proto_info",
+        77 => "ordered_table",
+        79 => "validity_cell",
+        80 => "weak_cell",
+        81 => "elements_double",
+        82 => "elements_dict",
+        83 => "fuse",
+        84 => "props_indexed",
+        85 => "bytes",
+        128 => "shape",
+        _ => return format!("kind{k}"),
+    };
+    n.to_string()
+}

+impl Drop for Counters {
+    fn drop(&mut self) {
+        let p = |s: String| eprintln!("ep9 census: {s}");
+        // cells by kind
+        let (mut tn, mut tb) = (0u64, 0u64);
+        let mut rows: Vec<(u64, u64, usize)> = (0..256)
+            .map(|k| {
+                let (n, b) = ep9_heap::census::get(k);
+                (b, n, k)
+            })
+            .filter(|r| r.1 > 0)
+            .collect();
+        rows.sort_by(|a, b| b.cmp(a));
+        for &(b, n, k) in &rows {
+            tn += n;
+            tb += b;
+            p(format!("kind {} id={k} cells={n} bytes={b}", kind_name(k)));
+        }
+        p(format!("kind TOTAL cells={tn} bytes={tb}"));
+        // SLOTS
+        p(format!(
+            "slots grow cells={} non_plain_object={}",
+            self.slots_n.get(),
+            self.slots_non_object.get()
+        ));
+        for (k, v) in self.slots_by_new_cap.borrow().iter() {
+            p(format!("slots new_cap={k} cells={v}"));
+        }
+        for (k, v) in self.slots_by_old_cap.borrow().iter() {
+            p(format!("slots old_cap={k} cells={v}"));
+        }
+        // classes
+        let classes = self.classes.borrow();
+        let mut cl: Vec<(&u64, &Class)> = classes.iter().collect();
+        cl.sort_by(|a, b| b.1.slots_cells.cmp(&a.1.slots_cells).then(b.1.adds.cmp(&a.1.adds)));
+        let (mut cov, mut all) = ([0u64; 3], 0u64);
+        for (_, c) in &cl {
+            all += c.slots_cells;
+            for i in 0..3 {
+                cov[i] += c.covered[i];
+            }
+        }
+        p(format!(
+            "slots plain_object cells={all} covered_k0={} covered_k1={} covered_k2={}",
+            cov[0], cov[1], cov[2]
+        ));
+        let (mut ex, mut adds) = ([0u64; 5], 0u64);
+        for (_, c) in &cl {
+            adds += c.adds;
+            for i in 0..5 {
+                ex[i] += c.extra[i];
+            }
+        }
+        p(format!(
+            "adds plain_object={adds} extra_e1={} e2={} e3={} e4={} e5plus={}",
+            ex[0], ex[1], ex[2], ex[3], ex[4]
+        ));
+        p(format!("named_load plain_object no_class={}", self.load_no_class.get()));
+        for (_, c) in &cl {
+            let bf: Vec<String> = c.by_f.iter().map(u64::to_string).collect();
+            let lf: Vec<String> = c.loads_by_f.iter().map(|x| format!("{}/{}", x[0], x[1])).collect();
+            p(format!(
+                "classf {} base={} names={} extra={} escaped={} bytecode={} today_cap=[{}] by_f=[{}] loads_by_f=[{}]",
+                c.chain.name,
+                c.obj_base,
+                c.chain.names.len(),
+                c.chain.extra,
+                c.chain.escaped,
+                c.chain.bytecode,
+                c.today_cap.iter().map(|(k, v)| format!("{k}:{v}")).collect::<Vec<_>>().join(" "),
+                bf.join(" "),
+                lf.join(" ")
+            ));
+        }
+        for (_, c) in cl.iter().take(40) {
+            let ch = &c.chain;
+            let mut names: Vec<&(String, u64)> = c.extra_names.values().collect();
+            names.sort_by(|a, b| b.1.cmp(&a.1));
+            let nm: Vec<String> = names
+                .iter()
+                .take(4)
+                .map(|(k, v)| format!("{k}:{v}"))
+                .collect();
+            let caps: Vec<String> = c.today_cap.iter().map(|(k, v)| format!("{k}:{v}")).collect();
+            p(format!(
+                "class {} bytecode={} levels={} own_names={} chain_names={} proto_extra={} escaped={} rule_cap_k0={} instances~{} adds={} max_f={} today_cap=[{}] slots_cells={} slots_bytes={} covered=[{} {} {}] extra=[{} {} {} {} {}] extra_names=[{}]",
+                ch.name,
+                ch.bytecode,
+                ch.levels,
+                ch.own_n,
+                ch.names.len(),
+                ch.extra,
+                ch.escaped,
+                rule_cap(ch, 0, 8),
+                c.first_adds,
+                c.adds,
+                c.max_f,
+                caps.join(" "),
+                c.slots_cells,
+                c.slots_bytes,
+                c.covered[0],
+                c.covered[1],
